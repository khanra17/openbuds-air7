cmake_minimum_required(VERSION 3.20)
project(openbuds_consumer LANGUAGES C)
set(OPENBUDS_SDK_DIR "" CACHE PATH "Extracted SDK; no Rust toolchain needed")
find_path(OPENBUDS_INCLUDE openbuds.h HINTS "${OPENBUDS_SDK_DIR}/include" REQUIRED NO_DEFAULT_PATH)
if(WIN32)
  find_library(OPENBUDS_LIBRARY NAMES openbuds.dll HINTS "${OPENBUDS_SDK_DIR}/lib" REQUIRED NO_DEFAULT_PATH)
else()
  find_library(OPENBUDS_LIBRARY NAMES openbuds HINTS "${OPENBUDS_SDK_DIR}/lib" REQUIRED NO_DEFAULT_PATH)
endif()
add_executable(abi_smoke abi_smoke.c)
target_compile_features(abi_smoke PRIVATE c_std_11)
target_include_directories(abi_smoke PRIVATE "${OPENBUDS_INCLUDE}")
target_link_libraries(abi_smoke PRIVATE "${OPENBUDS_LIBRARY}")
if(WIN32)
  add_custom_command(TARGET abi_smoke POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${OPENBUDS_SDK_DIR}/bin/openbuds.dll" "$<TARGET_FILE_DIR:abi_smoke>")
else()
  set_target_properties(abi_smoke PROPERTIES BUILD_RPATH "${OPENBUDS_SDK_DIR}/lib")
endif()
enable_testing()
add_test(NAME c_abi COMMAND abi_smoke)
